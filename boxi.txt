    <!-- 交互脚本 -->
    <script>
        const DEFAULT_PHOTO_URL = "https://images.unsplash.com/photo-1534528741775-53994a69daeb?auto=format&fit=crop&q=80&w=400";
        const STORAGE_KEY = "lin_pohsi_profile_photo";

        // 頁面載入時檢查是否有儲存的自訂照片
        window.addEventListener('DOMContentLoaded', () => {
            document.getElementById('year').textContent = new Date().getFullYear();
            try {
                const savedPhoto = localStorage.getItem(STORAGE_KEY);
                if (savedPhoto) {
                    document.getElementById('profile-img').src = savedPhoto;
                    document.getElementById('reset-photo-btn').classList.remove('hidden');
                }
            } catch (e) {
                console.warn('無法讀取本地儲存的照片:', e);
            }
        });

        // 處理本地照片選擇與上傳預覽
        function handlePhotoUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            if (!file.type.startsWith('image/')) {
                showToast('請選擇有效的圖片檔案！');
                return;
            }

            const reader = new FileReader();
            reader.onload = function(e) {
                const base64Image = e.target.result;
                document.getElementById('profile-img').src = base64Image;
                document.getElementById('reset-photo-btn').classList.remove('hidden');

                try {
                    localStorage.setItem(STORAGE_KEY, base64Image);
                    showToast('照片已成功更新！');
                } catch (err) {
                    // 若圖片檔案過大超過 localStorage 限制，仍保留當次瀏覽預覽
                    showToast('照片已預覽（檔案較大，重新整理後將恢復預設）');
                }
            };
            reader.readAsDataURL(file);
            event.target.value = ''; // 清空 input 以便重複選擇同一檔案
        }

        // 重設回預設照片
        function resetPhoto() {
            document.getElementById('profile-img').src = DEFAULT_PHOTO_URL;
            document.getElementById('reset-photo-btn').classList.add('hidden');
            try {
                localStorage.removeItem(STORAGE_KEY);
            } catch (e) {}
            showToast('已恢復為預設照片');
        }

        // 复制到剪贴板功能
        function copyToClipboard(text, successMessage) {
            // 使用标准的 execCommand 作为兼容 fallback
            const tempInput = document.createElement('input');
            tempInput.value = text;
            document.body.appendChild(tempInput);
            tempInput.select();
            
            try {
                document.execCommand('copy');
                showToast(successMessage);
            } catch (err) {
                showToast('複製失敗，請手動複製');
            }
            document.body.removeChild(tempInput);
        }

        // 显示 Toast 提示框
        function showToast(message) {
            const container = document.getElementById('toast-container');
            const toast = document.createElement('div');
            toast.className = 'bg-slate-900 text-white text-xs font-medium px-4 py-3 rounded-xl shadow-xl flex items-center gap-2 animate-toast pointer-events-auto';
            toast.innerHTML = `<i class="fa-solid fa-circle-check text-emerald-400"></i> ${message}`;
            
            container.appendChild(toast);

            setTimeout(() => {
                toast.remove();
            }, 2500);
        }
    </script>
